cmake_minimum_required(VERSION 3.20)
project(libreloop C)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(CTest)
add_library(engine src/engine.c src/arrangement.c src/sampler.c src/waveform.c src/navigation.c)
target_include_directories(engine PUBLIC src)
target_link_libraries(engine PUBLIC m)
target_compile_options(engine PRIVATE -Wall -Wextra -Wpedantic)
add_library(windows src/windows.c)
target_include_directories(windows PUBLIC src)
target_link_libraries(windows PUBLIC m)
target_compile_options(windows PRIVATE -Wall -Wextra -Wpedantic)
function(libreloop_test name library)
  if(BUILD_TESTING)
    add_executable(${name}_test tests/${name}_test.c)
    target_link_libraries(${name}_test PRIVATE ${library})
    add_test(NAME ${name} COMMAND ${name}_test)
  endif()
endfunction()
libreloop_test(engine engine)
libreloop_test(arrangement engine)
libreloop_test(audio_clip engine)
libreloop_test(clip_trim engine)
libreloop_test(sampler engine)
libreloop_test(stereo engine)
libreloop_test(navigation engine)
libreloop_test(waveform engine)
libreloop_test(windows windows)
add_executable(make_samples EXCLUDE_FROM_ALL tools/make_samples.c)
target_link_libraries(make_samples PRIVATE engine)
option(BUILD_GUI "Build the desktop application" ON)
if(BUILD_GUI)
  if(APPLE)
    enable_language(OBJC)
  endif()
  include(FetchContent)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(CUSTOMIZE_BUILD ON CACHE BOOL "" FORCE)
  set(SUPPORT_MODULE_RAUDIO OFF CACHE BOOL "" FORCE)
  set(SUPPORT_MODULE_RMODELS OFF CACHE BOOL "" FORCE)
  set(SUPPORT_CAMERA_SYSTEM OFF CACHE BOOL "" FORCE)
  set(SUPPORT_GIF_RECORDING OFF CACHE BOOL "" FORCE)
  set(SUPPORT_PARTIALBUSY_WAIT_LOOP OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(raylib
    URL https://codeload.github.com/raysan5/raylib/tar.gz/refs/tags/5.5
    URL_HASH SHA256=aea98ecf5bc5c5e0b789a76de0083a21a70457050ea4cc2aec7566935f5e258e)
  FetchContent_MakeAvailable(raylib)
  add_library(audio src/audio.c)
  find_package(Threads REQUIRED)
  target_include_directories(audio SYSTEM PRIVATE ${raylib_SOURCE_DIR}/src/external)
  target_link_libraries(audio PUBLIC engine Threads::Threads ${CMAKE_DL_LIBS})
  if(APPLE)
    target_compile_definitions(audio PRIVATE MA_NO_RUNTIME_LINKING)
    find_library(CORE_FOUNDATION CoreFoundation REQUIRED)
    find_library(CORE_AUDIO CoreAudio REQUIRED)
    find_library(AUDIO_TOOLBOX AudioToolbox REQUIRED)
    target_link_libraries(audio PUBLIC ${CORE_FOUNDATION} ${CORE_AUDIO} ${AUDIO_TOOLBOX})
  endif()
  add_library(browser src/browser.c)
  target_include_directories(browser PUBLIC src)
  target_link_libraries(browser PUBLIC raylib)
  target_compile_options(browser PRIVATE -Wall -Wextra)
  add_executable(libreloop src/main.c src/theme.c)
  if(APPLE)
    target_sources(libreloop PRIVATE src/macos_input.m)
    set_source_files_properties(src/macos_input.m PROPERTIES COMPILE_OPTIONS "-fblocks")
    set_target_properties(libreloop PROPERTIES MACOSX_BUNDLE TRUE
      MACOSX_BUNDLE_BUNDLE_NAME LibreLoop
      MACOSX_BUNDLE_GUI_IDENTIFIER org.libreloop.app
      MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Info.plist.in")
    set(bundle_samples
      "${CMAKE_CURRENT_SOURCE_DIR}/samples/drums/Kick.wav"
      "${CMAKE_CURRENT_SOURCE_DIR}/samples/drums/Snare.wav"
      "${CMAKE_CURRENT_SOURCE_DIR}/samples/drums/Hat.wav"
      "${CMAKE_CURRENT_SOURCE_DIR}/samples/melodic/Tone.wav")
    foreach(sample IN LISTS bundle_samples)
      file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}" "${sample}")
      get_filename_component(directory "${relative}" DIRECTORY)
      set_source_files_properties("${sample}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/${directory}")
    endforeach()
    set(bundle_font "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/LiberationSans-Regular.ttf")
    set_source_files_properties("${bundle_font}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/fonts")
    set(bundle_notices "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE"
      "${CMAKE_CURRENT_SOURCE_DIR}/docs/third-party.md")
    file(GLOB dependency_notices "${CMAKE_CURRENT_SOURCE_DIR}/docs/licenses/*.txt")
    list(APPEND bundle_notices ${dependency_notices})
    set_source_files_properties(${bundle_notices} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/licenses")
    set(font_notice "${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/LICENSE")
    set_source_files_properties("${font_notice}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/licenses/fonts")
    target_sources(libreloop PRIVATE ${bundle_samples} "${bundle_font}" ${bundle_notices} "${font_notice}")
    install(TARGETS libreloop BUNDLE DESTINATION .)
  endif()
  target_compile_definitions(libreloop PRIVATE LIBRELOOP_SAMPLES="${CMAKE_CURRENT_SOURCE_DIR}/samples"
    LIBRELOOP_FONT="${CMAKE_CURRENT_SOURCE_DIR}/assets/fonts/LiberationSans-Regular.ttf")
  target_link_libraries(libreloop PRIVATE audio browser windows glfw)
  target_compile_options(libreloop PRIVATE -Wall -Wextra)
  libreloop_test(decoder audio)
  libreloop_test(audio audio)

    target_include_directories(audio_test SYSTEM PRIVATE ${raylib_SOURCE_DIR}/src/external)
    if(APPLE)
      target_compile_definitions(audio_test PRIVATE MA_NO_RUNTIME_LINKING)
    endif()
  endif()
  libreloop_test(browser browser)
endif()
